### 5.2 方法差异的实质：k 与分布要求的此消彼长

import re, sys
from pathlib import Path
LEAN_FILE = Path(__file__).resolve().parent / "PrimeGaps186" / "PrimeGaps186.lean"
EXPECTED_PRIME_CHECK_BOUND = 40

def parse_tuple_from_lean(path):
    text = path.read_text(encoding="utf-8")
    m = re.search(r"def admissibleTuple : Finset ℕ :=\s*\{([^}]*)\}", text)
    if not m: raise RuntimeError("definition not found")
    return [int(tok) for tok in re.findall(r"\d+", m.group(1))]

def sieve_primes(n):
    is_p = [True]*(n+1); is_p[0] = is_p[1] = False
    for i in range(2, int(n**0.5)+1):
        if is_p[i]:
            for j in range(i*i, n+1, i): is_p[j] = False
    return [i for i, v in enumerate(is_p) if v]

    H = parse_tuple_from_lean(LEAN_FILE)
    card, uniq = len(H), len(set(H))
    ok &= (card == 40 and uniq == 40)                      # [1] 基数
    ok &= (max(H) - min(H) == 186)                          # [2] 直径
